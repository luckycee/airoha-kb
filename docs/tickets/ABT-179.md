 send_buffer = (uint8_t *)pvPortMalloc(5 + test_length);
 
#if 0
 if (send_buffer == NULL) \{
 LOG_MSGID_I(AIR, "[BLE][AIR] notification alloc buffer fail", 0);
 return BT_STATUS_FAIL;
 }
 
#endif
 bt_gattc_charc_value_notification_indication_t *notification = (bt_gattc_charc_value_notification_indication_t *)send_buffer;
 notification->attribute_value_length = 3 + test_length;
 notification->att_req.opcode = BT_ATT_OPCODE_HANDLE_VALUE_NOTIFICATION;
 notification->att_req.handle = 85;
 memcpy(notification->att_req.attribute_value, (void *)(&battery_level), sizeof(uint8_t));
 
 printf("[人名5] bt_gatts_send_charc_value_notification_indication 5 ", 0);
 status = bt_gatts_send_charc_value_notification_indication(5, notification);
 if (status != BT_STATUS_SUCCESS) \{

 }

}
